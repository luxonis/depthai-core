cmake_minimum_required(VERSION 3.20)
project(apriltag_windows_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(apriltag CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(spdlog CONFIG REQUIRED)
find_package(Eigen3 CONFIG REQUIRED)
add_executable(apriltag_smoke smoke.cpp)
target_compile_definitions(apriltag_smoke PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(apriltag_smoke PRIVATE apriltag::apriltag)
foreach(probe IN ITEMS sdk_apriltag sdk_apriltag_math_fix)
  add_library(${probe} OBJECT sdk/src/pipeline/node/AprilTag.cpp)
  target_include_directories(${probe} PRIVATE sdk/include sdk/include/depthai sdk/include/3rdparty sdk/src xlink/include libnop/include)
  target_compile_definitions(${probe} PRIVATE DEPTHAI_HAS_APRIL_TAG NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(${probe} PRIVATE apriltag::apriltag nlohmann_json::nlohmann_json spdlog::spdlog Eigen3::Eigen)
endforeach()
target_compile_definitions(sdk_apriltag_math_fix PRIVATE _USE_MATH_DEFINES)
