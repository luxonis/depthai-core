cmake_minimum_required(VERSION 3.20)
project(apriltag_sdk_integration LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# The minimal SDK configuration independently needs MSVC's math constants.
add_compile_definitions(_USE_MATH_DEFINES)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/sdk/resources")
add_subdirectory(../sdk sdk)
# This is a functional probe; skip expensive whole-library optimizations.
if(MSVC)
  target_compile_options(depthai-core PRIVATE /Od /Ob0)
endif()
find_package(apriltag CONFIG REQUIRED)
add_executable(apriltag_node_smoke node.cpp)
target_compile_definitions(apriltag_node_smoke PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(apriltag_node_smoke PRIVATE depthai::core apriltag::apriltag)
